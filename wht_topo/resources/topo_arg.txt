--bead-height 15.0

# 최소 비드 폭 최종 목표값 (mm). 밀도 필터 반경으로 작용

# 전략: 초기 넓은 필터로 뼈대 형성 → 후반 좁은 필터로 비드 선 얇아짐
# 비활성화(초기=최종, ramp 없음): -1 또는 주석 처리

# 비드 점유 면적 비율 제약 (0~1). 전체 면적 대비 비드가 차지할 수 있는 최대 비율

--bead-area 0.50

# 초반 자유도 확보 → 위상 패턴 형성 후 점진적 면적 제한
# 0: 즉시 목표값 적용. 권장: 5~15 (bead-area가 작을수록 길게)
# --bead-area-ramp 5


# =============================================================================
# [4] 목적함수 옵션
# =============================================================================

# 케이스별 정규화 및 가중치 변동
# 0   = 비활성 (원시 컴플라이언스 합산)
# 1.0 = 정규화 활성 (각 케이스를 초기 컴플라이언스로 나눔, 변동 없음)
# 1.4 = 정규화 + 케이스별 ±0.4 랜덤 가중치 부여 (다양성 유도)
# 권장: 1.0 (안정), 다중 설계 탐색 시 1.2~1.5
--normalize-obj 1.0

# 목적함수 집계 방식
# sum     = 모든 케이스의 가중합 (수렴 안정)
# max     = KS softmax 최악 케이스 집중 (강건 설계)
# sum+max = 0.5·sum + 0.5·softmax (균형, 기본 권장)
--obj-type max

# KS softmax 온도 파라미터 (--obj-type max 또는 sum+max 시 적용)
# 낮을수록 (1~3): 케이스 평균에 가까운 완만한 max 근사
# 기본 (10):      실용적인 최악케이스 집중
# 높을수록 (50+): hard-max 근사 → 수렴 불안정 주의
--obj-alpha 10.0

# 고유진동수 패널티: W·(max(0, f_target - f1) / f_target)²
# [가중치, 목표진동수(Hz)]
# [0.0, 0.0] = 비활성 (기본)
# 예: [5.0, 50.0] → 1차 고유진동수가 50Hz 미만이면 패널티 부과
--freq-penalty 0.0 0.0


# =============================================================================
# [5] 다중 설계 탐색 (Multi-Design Exploration)
# =============================================================================

# 생성할 설계 수. 1 = 단일 최적 설계 (기본)
# >1이면 첫 번째 수렴 후 반발 패널티를 추가하여 다른 설계 탐색
# 출력: _d01.k, _d02.k, ... 파일 생성
--n-designs 1

# 반발 패널티 강도 λ. 클수록 이전 설계에서 멀리 떨어진 설계 유도
# f_total = C + λ·Σ exp(-||x - x*_k||² / 2σ²)
# 권장: 0.1~0.5 (너무 크면 컴플라이언스 성능 저하)
--diversity-weight 0.3

# 반발 영향 범위 σ (설계 변수 공간 단위, 0~1 정규화 기준)
# 클수록 이전 설계 주변 넓은 영역 회피
# 권장: 0.2~0.5
--diversity-sigma 0.3

# 설계 재시작 시 초기 노이즈 강도 (기본: 0.15)
# 이전 수렴 설계에 randn * noise를 더해 새 출발점 설정
# 너무 작으면 같은 곳으로 수렴, 너무 크면 무작위 탐색
--diversity-noise 0.15

# 인라인 다형성: 지정 iter부터 반발 패널티 즉시 적용 (기본: -1=비활성)
# -1      : --n-designs 방식 (수렴 후 재시작)
# N >= 0  : 단일 루프 내 iter N에 현재 x 스냅샷 등록 후 패널티 ramp-in
# 전략: 초기 자유 탐색 → N iter 이후 현재 형태에서 멀어지는 방향 유도
# 권장: 전체 iters의 50~70%. 예) --iters 40 → --diversity-start-iter 25
--diversity-start-iter 10


# =============================================================================
# [6] 병렬 해석 옵션
# =============================================================================

# 하중 케이스 병렬 해석 스레드 수 (기본: 4)
# 케이스 수보다 크게 지정해도 케이스 수로 자동 제한
# BLAS 멀티스레드와 충돌 방지를 위해 CPU 코어 수의 절반 권장
# (예: 12코어 CPU → 4~6 권장)
--n-workers 4


# =============================================================================
# [7] 비드 형상 및 제조 제약
# =============================================================================

# 좌우(X축) 대칭 활성화 (기본: 활성)
# 비활성화 시: --no-sym-x 로 대체
--sym-x

# 비드 자동 연결 최대 갭 (mm)
# 0   = 비활성 (단절 비드 허용)
# >0  = 해당 갭 이하의 단절 비드를 Morphological Closing으로 연결
#       알고리즘: Dilation(gap/spacing 회) → Erosion(gap/spacing 회)
# 권장: mesh-xy × 3~5배 (기본: 120mm = mesh-xy 30mm × 4)
--bead-connect 150.0

# 비드 연결 알고리즘 선택
# closing  = Morphological Closing (기본, 빠름)
# mst      = Minimum Spanning Tree (긴 단절 비드 연결에 강함)
# geodesic = 측지선 경로 연결 (자연스러운 곡선 연결, 권장)
# hybrid   = geodesic + closing 혼합
--bead-connect-alg geodesic

# 비드 이산화 단계 수
# 0  = 연속 최적화 결과 그대로 (discrete projection 없음)
# 1  = 이진: {0, h_max}
# 2  = 3단계: {0, h_max/2, h_max}
# N  = N+1 단계 균등 분할: {0, h_max/N, ..., h_max}
--height-steps 2

# 양방향 비드 허용 (기본: 비활성 = 단방향)
# 활성화 시: x=0.5 → 높이 0(비드 없음), x=0 → -h_max, x=1 → +h_max
# steps=1 → {-h_max, 0, +h_max} (3레벨), steps=2 → 5레벨
# colorbar 자동 전환: coolwarm 발산형 ±h_max
# 활성화하려면 아래 주석 해제:
# --bead-bidirectional

# 공간 밀도 필터 커널 유형
# linear   = 선형 hat 커널 w = rmin - d (기본, 계산 빠름)
# gaussian = 가우시안 커널 w = exp(-d²/2σ²), σ=rmin/3 (부드러운 덩어리 형태)
--filter-type linear

# Heaviside projection beta 최대값 (기본: 0 = 비활성)
# 이터레이션 진행에 따라 beta: 1 → proj_beta_max 선형 증가
# H_β(x; η=0.5) = [tanh(βη) + tanh(β(x-η))] / [tanh(βη) + tanh(β(1-η))]
# 0    = 비활성 (연속값 유지)
# 8~16 = 부드러운 경계 (초기 탐색 단계 권장)
# 32   = 표준 (수렴 안정)
# 64+  = hard 경계 (0/1 이진에 가까움, 수렴 불안정 주의)
--projection 32

# 비드 돌출 방향 벡터 (쉼표 구분 x,y,z)
# 0,0,-1 = 아래 방향 (기본, 트레이 바닥 아래로 돌출)
# 0,0,1  = 위 방향
--draw-dir 0,0,-1

# 비드 배제 사각형 영역 (중심X,중심Y,가로W,세로H, 단위 mm)
# 여러 영역 지정: --exclude-rect 500,300,100,80 --exclude-rect 200,400,60,60
# 미사용 시 주석 처리
# --exclude-rect 500,300,100,80

# 비드 배제 다각형 영역 (꼭짓점 X1,Y1,X2,Y2,... 나열, mm)
# 여러 영역: --exclude-poly 100,200,150,200,150,250,100,250
# 미사용 시 주석 처리
# --exclude-poly 100,200,150,200,150,250,100,250


# =============================================================================
# [8] CSV 단독 동적 응답 해석 (모드 B, 최적화 생략)
# 주의: --pos-data 지정 시 최적화를 건너뛰고 동적 해석만 실행됨
# =============================================================================

# CSV 경로 (지정 시 동적 해석 전용 모드로 전환)
# --pos-data path/to/motion.csv

# 분석 시작 시점 (s). 기본: CSV 헤더 start_time 자동 사용
# --t-start 1.0

# 분석 종료 시점 (s). 기본: CSV 마지막 프레임
# --t-end 3.0

# 적분 시간 스텝 (s). ESL 모드에도 동일 적용
# 기본값 0.005s = 5ms (200Hz 해상도). 필요 시 dt ≤ 1/(10×f_max) 권장
--dt 0.005

# Rayleigh 감쇠비 ζ. 임계 감쇠의 비율 (0~1)
# 0.01~0.05: 금속 구조물 일반 범위, 0.02 = 2% 권장
--zeta 0.02

# 코너 노드 탐색 반경 (mm). 자동 코너 탐색 실패 시 fallback 반경
--corner-r 150.0


# =============================================================================
# [9] 동적 ESL 통합 최적화 (모드 C/D)
# =============================================================================

# ESL(등가 정적 하중) 시나리오 CSV 파일 경로 및 시간 범위
# 형식: CSV경로  또는  CSV경로,시작(s)  또는  CSV경로,시작(s),종료(s)
# 복수 시나리오: 줄마다 --dynamic-opts 반복
# 미사용 시 주석 처리
# --dynamic-opts chassis_corners.csv,0.1190,2.0010

# 관성 하중(-ma) 인가 여부
# 지정 시: 각 노드의 질량 × 가속도를 역방향으로 적용 (d'Alembert 관성력)
# 미사용 (기본): 외력만 적용
# --add-inertia

# ESL 추출 개수 (상위 N개 하중 상태 선택, 누적 SE 기준 윈도우 선정)
# 클수록 더 많은 동적 하중 상태를 정적 케이스로 변환 → 정밀도 향상, 계산 증가
--n-top 5

# 시간 이력 분할 윈도우 수. 충격 구간(G 기준 ×3 확장) 내에서 분할
# 클수록 시간 해상도 향상. n_top ≤ n_windows 권장
--n-windows 30

# ESL 추출용 동적 해석 저장 스냅샷 수
# n_windows*2 미만이면 자동으로 n_windows*2 로 상향
--n-save-esl 30

# 충격 구간 감지 임계 가속도 [mm/s²]
# Kabsch body-frame 가속도 크기가 이 값을 초과하는 구간을 충격 구간으로 판정
# 해당 구간의 지속시간 ×3 전후를 ESL 윈도우 탐색 범위로 사용
# 기본: 24517 ≈ 2.5g  (9806.65 mm/s² × 2.5)
--contact-threshold 24516.6

# 시간분할 ESL 스냅샷 가중치. CSV별 복수 지정 가능
# 예: --w-dynamic 1.0 0.8  → 첫 번째 CSV 가중치 1.0, 두 번째 0.8
--w-dynamic 1.0

# 요소별 최대 변형에너지 피크 ESL 가중치 (0.0 = 비활성)
# >0이면 최대 변형에너지 발생 시점의 하중 상태 별도 추가
--w-peak 0.0

# 매 이터레이션 ESL 재추출 여부 (기본: 활성)
# 활성(기본): 설계 변수 갱신마다 동해석 재실행 → 정확한 ESL
# --no-iterative-esl: 1회만 추출 후 고정 → 계산 빠름, 정밀도 낮음
--iterative-esl

# 복수 CSV 시나리오 동시 실행 프로세스 수 (기본: 1 = 순차)
# 2 이상 지정 시 ProcessPoolExecutor로 N개 시나리오를 동시 실행
--parallel-scenarios 1

# 반복 ESL 재추출 스킵 임계값 (기본: 0.0 = 항상 재실행)
# 이전 이터레이션 대비 Δh_rms < TOL 이면 동해석 생략하고 직전 ESL 재사용
--esl-skip-tol 0.0

# 모달 중첩법 사용 모드 수 (기본: 0 = 직접 Newmark-β 적분)
# 0  = 직접 Newmark-β 적분 (정확)
# >0 = 모달 중첩법 (수십~수백 배 빠름, 권장: 20~50)
--n-modes 0

# JAX 가속 솔버 사용 여부 (기본: scipy)
# n_free > 10000 이면 JAX JIT 컴파일 시간 과도 → scipy 권장
# --jax-dynamic

# 글로벌 Z 궤적 직접 사용 여부 (현재 미사용)
# --use-global-z

# 정적 하중 케이스 전체 비활성 (동적 ESL만 사용할 때)
# --no-static


# =============================================================================
# [10] 정적 하중 케이스 설정 (가중치, 하중N)
# 형식: W F_N  (가중치 하중크기, 음수=압축/아래방향)
# W=0 으로 설정하면 해당 케이스 비활성화
# --w-basic-weight 값이 0이 아니면 F_N은 자동 계산(자중 기반)으로 대체됨
# =============================================================================

# 자중 기반 하중 스케일 (기본: 1.0)
# 0이 아닌 값: 총 질량 × 9806mm/s² × SCALE 을 각 하중값으로 자동 대체
# 0: --w-xxx F_N 에 직접 입력한 값을 그대로 사용
